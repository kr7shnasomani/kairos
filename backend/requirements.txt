# --- Observability ---
# OTEL core + instrumentation move as a matched pair (1.x <-> 0.xxbN). Upgrade all six together.
#
# 1.24.0/0.45b0 -> 1.28.0/0.49b0 on 2026-08-23. The driver was `setuptools`, not OTEL:
# `opentelemetry.instrumentation.dependencies` imported `pkg_resources`, which setuptools 78+
# removes — so the `setuptools` cap in the Testing section was load-bearing, not stale as
# previously assumed, and lifting it alone crashed `api.main` on import.
# 0.49b0 is the first release that dropped `pkg_resources`, which is what unpins setuptools.
opentelemetry-api==1.28.0
opentelemetry-sdk==1.28.0
opentelemetry-exporter-otlp-proto-http==1.28.0
opentelemetry-instrumentation-fastapi==0.49b0
opentelemetry-instrumentation-redis==0.49b0
opentelemetry-instrumentation-httpx==0.49b0

# Transitive via qdrant-client, pinned explicitly because it is the linchpin of the protobuf
# constraint. qdrant-client asks only for `grpcio-tools>=1.41.0`, and pip was resolving to 1.62.3,
# whose `protobuf<5.0dev` cap held protobuf at 4.25.9 (a known advisory) and blocked any OTEL that
# needs protobuf 5. Raising THIS one package moves protobuf to 5.x with qdrant-client and
# temporalio untouched — temporalio declares `protobuf>=3.20` with no upper bound.
# Without the pin a future resolve can silently pick an old grpcio-tools and re-break the chain.
grpcio-tools>=1.66
